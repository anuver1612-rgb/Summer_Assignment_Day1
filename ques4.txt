import java.util.*;
class q4
{
    public static void main(String args[])
    {
        Scanner sc=new Scanner(System.in);
        System.out.println("Enter the number:");
        int n=sc.nextInt();
        int c=0,i;
        for(i=n;i!=0;i=i/10)
        {
            c++;
        }
        System.out.println("The number of digits = "+c);
    }
}
