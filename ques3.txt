import java.util.*;
class q3
{
    public static void main(String args[])
    {
        System.out.println("Enter the number:");
        Scanner sc = new Scanner(System.in);
        int n=sc.nextInt();
        int f=1,i;
        for(i=1;i<=n;i++)
        {
            f=f*i;
        }
        System.out.println("Factorial:"+f);
    }
}
